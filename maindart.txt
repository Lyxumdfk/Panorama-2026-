class Esp32PaymentResult {
  final String status;
  final String card;
  final String uid;
  final double amount;
  final double finalAmount;
  final double discountPercent;
  final double discountAmount;

  const Esp32PaymentResult({
    required this.status,
    this.card = '',
    this.uid = '',
    this.amount = 0,
    this.finalAmount = 0,
    this.discountPercent = 0,
    this.discountAmount = 0,
  });

  bool get approved => status == 'approved';
  bool get hasFlexiDiscount => card.toLowerCase() == 'flexi' && discountPercent > 0;

  factory Esp32PaymentResult.fromJson(Map<String, dynamic> json) {
    double readDouble(String key) {
      final value = json[key];
      if (value is num) return value.toDouble();
      return double.tryParse(value?.toString() ?? '') ?? 0;
    }

    return Esp32PaymentResult(
      status: (json['status'] ?? 'unknown').toString(),
      card: (json['card'] ?? '').toString(),
      uid: (json['uid'] ?? '').toString(),
      amount: readDouble('amount'),
      finalAmount: readDouble('finalAmount'),
      discountPercent: readDouble('discountPercent'),
      discountAmount: readDouble('discountAmount'),
    );
  }
}

Future<bool> startEsp32Payment({
  required double amount,
  required int orderId,
  required String tableName,
}) async {
  try {
    final response = await http.post(
      Uri.parse('$esp32BaseUrl/start-payment'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({
        'amount': amount,
        'orderId': orderId,
        'table': tableName,
      }),
    );

    if (response.statusCode != 200) return false;

    final data = jsonDecode(response.body);
    return data['success'] == true;
  } catch (e) {
    debugPrint('ESP32 start payment error: $e');
    return false;
  }
}

Future<Esp32PaymentResult> getEsp32PaymentResult() async {
  try {
    final response = await http.get(
      Uri.parse('$esp32BaseUrl/payment-status'),
    );

    if (response.statusCode != 200) {
      return const Esp32PaymentResult(status: 'error');
    }

    final data = jsonDecode(response.body);
    if (data is Map<String, dynamic>) {
      return Esp32PaymentResult.fromJson(data);
    }

    return const Esp32PaymentResult(status: 'unknown');
  } catch (e) {
    debugPrint('ESP32 payment status error: $e');
    return const Esp32PaymentResult(status: 'error');
  }
}

Future<String> getEsp32PaymentStatus() async {
  final result = await getEsp32PaymentResult();
  return result.status;
}

Future<Esp32PaymentResult> waitForEsp32PaymentResult({
  Duration timeout = const Duration(seconds: 90),
}) async {
  final endTime = DateTime.now().add(timeout);

  while (DateTime.now().isBefore(endTime)) {
    await Future.delayed(const Duration(seconds: 1));

    final result = await getEsp32PaymentResult();

    if (result.status == 'approved') return result;

    if (result.status == 'denied' ||
        result.status == 'cancelled' ||
        result.status == 'error') {
      return result;
    }
  }

  return const Esp32PaymentResult(status: 'timeout');
}

Future<bool> waitForEsp32Payment({
  Duration timeout = const Duration(seconds: 90),
}) async {
  final result = await waitForEsp32PaymentResult(timeout: timeout);
  return result.approved;
}

enum AppNoticeType { success, error, warning, info }

Color noticeColor(AppNoticeType type) {
  switch (type) {
    case AppNoticeType.success:
      return const Color(0xFF009B6B);
    case AppNoticeType.error:
      return const Color(0xFFE00035);
    case AppNoticeType.warning:
      return const Color(0xFFFF8A00);
    case AppNoticeType.info:
      return const Color(0xFF006CFF);
  }
}

class FlexiRepository {
  Future<AppUserModel?> tryUserLogin(String pin, {int? role}) async {
    var query = supabase
        .from('users')
        .select('id, name, role, pinhash, pinsalt, pin_code, isactive')
        .eq('isactive', true);

    if (role != null) {
      query = query.eq('role', role);
    }

    final data = await query.order('id');

    for (final row in (data as List)) {
      final user = AppUserModel.fromMap(row as Map<String, dynamic>);
      if (user.pinCode == pin || user.pinHash == pin) {
        return user;
      }
    }

    return null;
  }

  Future<AppUserModel?> tryAdminLogin(String pin) async {
    return tryUserLogin(pin, role: 1);
  }

  Future<List<AppUserModel>> getUsers() async {
    final data = await supabase
        .from('users')
        .select('id, name, role, pinhash, pinsalt, pin_code, isactive')
        .order('id');

    return (data as List)
        .map((e) => AppUserModel.fromMap(e as Map<String, dynamic>))
        .toList();
  }

  Future<void> createUser(String name, int role, String pin) async {
    await supabase.from('users').insert({
      'name': name.trim(),
      'role': role,
      'pin_code': pin.trim(),
      'pinhash': pin.trim(),
      'pinsalt': '',
      'isactive': true,
    });
  }

  Future<void> updateUser(int id, String name, int role, bool isActive) async {
    await supabase.from('users').update({
      'name': name.trim(),
      'role': role,
      'isactive': isActive,
    }).eq('id', id);
  }

  Future<void> updateUserPin(int id, String pin) async {
    await supabase.from('users').update({
      'pin_code': pin.trim(),
      'pinhash': pin.trim(),
      'pinsalt': '',
    }).eq('id', id);
  }

  Future<void> deleteUser(int id) async {
    await supabase.from('users').delete().eq('id', id);
  }

  Future<List<ZoneModel>> getZones() async {
    final data = await supabase
        .from('zones')
        .select('id, name, isactive')
        .order('name');

    return (data as List)
        .map((e) => ZoneModel.fromMap(e as Map<String, dynamic>))

  Future<Map<String, dynamic>?> getLatestOpenOrderByTable(int tableId) async {
    final data = await supabase
        .from('orders')
        .select('id, table_id, user_id, created_by_user_id, status, total_price, subtotal')
        .eq('table_id', tableId)
        .eq('status', 'open')
        .order('id', ascending: false)
        .limit(1);

    final rows = (data as List).cast<Map<String, dynamic>>();
    return rows.isEmpty ? null : rows.first;
  }

  Future<void> verifyTableAccess(int tableId, int userId) async {
    final data = await supabase
        .from('tables')
        .select('id, status, owner_user_id')
        .eq('id', tableId)
        .limit(1);

    final rows = (data as List).cast<Map<String, dynamic>>();
    if (rows.isEmpty) throw Exception('Масата не е намерена.');

    final table = rows.first;
    final status = _readInt(table, ['status']);
    final ownerId = _readInt(table, ['owner_user_id']);

    if (status == 1 && ownerId > 0 && ownerId != userId) {
      throw Exception('Тази маса е заета от друг сервитьор. Само той може да влиза, да поръчва и да приключва сметката.');
    }

    if (status == 1 && ownerId <= 0) {
      throw Exception('Тази маса е заета, но няма записан сервитьор. Сметката е заключена.');
    }
  }

  Future<int> ensureOpenOrder(int tableId, int userId) async {
    await verifyTableAccess(tableId, userId);

    final openOrder = await getLatestOpenOrderByTable(tableId);
    if (openOrder != null) {
      final ownerId = _readInt(openOrder, ['created_by_user_id', 'user_id']);
      if (ownerId != userId) {
        throw Exception('Тази маса има отворена сметка от друг сервитьор. Нямаш право да я пипаш.');
      }
      return _readInt(openOrder, ['id']);
    }

    final inserted = await supabase.from('orders').insert({
      'user_id': userId,
      'created_by_user_id': userId,
      'table_id': tableId,
      'created_at': DateTime.now().toUtc().toIso8601String(),
      'status': 'open',
      'total_price': 0,
      'subtotal': 0,
      'paid_total': 0,
    }).select('id').single();

    return _readInt(inserted as Map<String, dynamic>, ['id']);
  }

  Future<double> refreshOrderTotal(int orderId) async {
    final data = await supabase
        .from('order_items')
        .select('quantity, unit_price')
        .eq('order_id', orderId);

    double total = 0;
    for (final row in (data as List)) {
      final map = row as Map<String, dynamic>;
      total += _readInt(map, ['quantity', 'qty']) * _readDouble(map, ['unit_price', 'price']);
    }

    await supabase.from('orders').update({
      'subtotal': total,
      'total_price': total,
    }).eq('id', orderId);

    return total;
  }

  Future<void> placeOrderItems({
    required int tableId,
    required int userId,
    required List<PosOrderLine> items,
  }) async {
    if (items.isEmpty) return;

    final orderId = await ensureOpenOrder(tableId, userId);

    final rows = items
        .where((x) => x.quantity != 0)
        .map((x) => {
              'order_id': orderId,
              'product_id': x.productId,
              'product_name': x.productName,
              'quantity': x.quantity,
              'unit_price': x.unitPrice,
              'ordered_at': DateTime.now().toUtc().toIso8601String(),
              'ordered_by_user_id': userId,
            })
        .toList();

    if (rows.isEmpty) return;

    await supabase.from('order_items').insert(rows);
    final total = await refreshOrderTotal(orderId);

    await supabase.from('tables').update({
      'status': 1,
      'owner_user_id': userId,
      'opened_at_utc': DateTime.now().toUtc().toIso8601String(),
      'current_total': total,
    }).eq('id', tableId);
  }

  Future<List<PosOrderLine>> getOpenOrderLinesForTable(int tableId, {int? userId}) async {
    if (userId != null) await verifyTableAccess(tableId, userId);

    final openOrder = await getLatestOpenOrderByTable(tableId);
    if (openOrder == null) return [];

    if (userId != null) {
      final ownerId = _readInt(openOrder, ['created_by_user_id', 'user_id']);
      if (ownerId != userId) {
        throw Exception('Тази маса има отворена сметка от друг сервитьор. Нямаш право да я отваряш.');
      }
    }

    final orderId = _readInt(openOrder, ['id']);
    final data = await supabase
        .from('order_items')
        .select('id, product_id, product_name, quantity, unit_price')
        .eq('order_id', orderId)
        .order('id');

    final grouped = <String, PosOrderLine>{};
    for (final row in (data as List)) {
      final map = row as Map<String, dynamic>;
      final productId = _readInt(map, ['product_id']);
      final productName = _readString(map, ['product_name'], fallback: 'Без име');
      final unitPrice = _readDouble(map, ['unit_price', 'price']);
      final quantity = _readInt(map, ['quantity', 'qty']);
      final key = '$productId|$productName|$unitPrice';
      final old = grouped[key];
      grouped[key] = PosOrderLine(
        productId: productId,
        productName: productName,
        unitPrice: unitPrice,
        quantity: (old?.quantity ?? 0) + quantity,
      );
    }

    return grouped.values.where((x) => x.quantity != 0).toList();
  }

  Future<void> closeOrder({
    required int tableId,
    required int userId,
    required String paymentMethod,
    double discountPercent = 0,
    double tipAmount = 0,
  }) async {
    await verifyTableAccess(tableId, userId);

    final openOrder = await getLatestOpenOrderByTable(tableId);
    if (openOrder == null) return;

    final orderId = _readInt(openOrder, ['id']);
    final ownerId = _readInt(openOrder, ['created_by_user_id', 'user_id']);
    if (ownerId != userId) {
      throw Exception('Само сервитьорът, който е отворил масата, може да приключи сметката.');
    }

    final subtotal = await refreshOrderTotal(orderId);
    final safeDiscount = discountPercent.clamp(0, 100).toDouble();
    final safeTip = tipAmount < 0 ? 0.0 : tipAmount;
    final discountAmount = subtotal * safeDiscount / 100.0;
    final finalTotal = (subtotal - discountAmount) + safeTip;

    await supabase.from('orders').update({
      'subtotal': subtotal,
      'discount_percent': safeDiscount,
      'discount_amount': discountAmount,
      'tip_amount': safeTip,
      'paid_total': finalTotal,
      'total_price': finalTotal,
      'payment_method': paymentMethod,
      'status': 'paid',
      'closed_at': DateTime.now().toUtc().toIso8601String(),
    }).eq('id', orderId);

    await supabase.from('tables').update({
      'status': 0,
      'owner_user_id': null,
      'opened_at_utc': null,
      'current_total': 0,
    }).eq('id', tableId);
  }

  Future<List<ActiveOrderItem>> getActiveItemsForTable(int tableId) async {
    final ordersData = await supabase
        .from('orders')
        .select('id, table_id, user_id, created_at, status')
        .eq('table_id', tableId)
        .eq('status', 'open')
        .order('created_at', ascending: false);

    final orders = (ordersData as List).cast<Map<String, dynamic>>();
    if (orders.isEmpty) return [];

    final orderIds = orders.map((e) => e['id'] as int).toList();
    final userIds = orders
        .map((e) => e['user_id'])
        .where((e) => e != null)
        .map((e) => e as int)
        .toSet()
        .toList();

    final Map<int, String> userNames = {};
    if (userIds.isNotEmpty) {
      final usersData = await supabase
          .from('users')
          .select('id, name')
          .inFilter('id', userIds);

      for (final row in (usersData as List)) {
        final map = row as Map<String, dynamic>;
        userNames[(map['id'] ?? 0) as int] = (map['name'] ?? '').toString();
      }
    }

    final itemsData = await supabase
        .from('order_items')
        .select('order_id, product_name, quantity, unit_price')
        .inFilter('order_id', orderIds);

    final items = <ActiveOrderItem>[];

    for (final row in (itemsData as List)) {
      final map = row as Map<String, dynamic>;
      final orderId = (map['order_id'] ?? 0) as int;
      final order = orders.firstWhere((e) => e['id'] == orderId);
      final createdAt = DateTime.tryParse((order['created_at'] ?? '').toString())?.toLocal();

      items.add(
        ActiveOrderItem(
          productName: (map['product_name'] ?? 'Без име').toString(),
          quantity: (map['quantity'] ?? 0) as int,
          unitPrice: ((map['unit_price'] ?? 0) as num).toDouble(),
          soldBy: userNames[(order['user_id'] ?? 0) as int] ?? 'Unknown',
          time: createdAt,
        ),
      );
    }

    return items;
  }

  int _readInt(Map<String, dynamic> map, List<String> keys) {
    for (final key in keys) {
      final value = map[key];
      if (value == null) continue;
      if (value is int) return value;
      if (value is num) return value.toInt();
      final parsed = int.tryParse(value.toString());
      if (parsed != null) return parsed;
    }
    return 0;
  }

  double _readDouble(Map<String, dynamic> map, List<String> keys) {
    for (final key in keys) {
      final value = map[key];
      if (value == null) continue;
      if (value is num) return value.toDouble();
      final parsed = double.tryParse(value.toString().replaceAll(',', '.'));
      if (parsed != null) return parsed;
    }
    return 0;
  }

  String _readString(
    Map<String, dynamic> map,
    List<String> keys, {
    String fallback = '',
  }) {
    for (final key in keys) {
      final value = map[key];
      if (value == null) continue;
      final text = value.toString();
      if (text.trim().isNotEmpty) return text;
    }
    return fallback;
  }

  Future<List<ReportSaleItem>> getDailyReport(DateTime day) async {
    final start = DateTime(day.year, day.month, day.day);
    final end = start.add(const Duration(days: 1));

    final ordersData = await supabase
        .from('orders')
        .select('id, user_id, created_by_user_id, created_at')
        .gte('created_at', start.toIso8601String())
        .lt('created_at', end.toIso8601String());

    final orders = (ordersData as List).cast<Map<String, dynamic>>();
    if (orders.isEmpty) return [];

    final orderIds = orders
        .map((e) => _readInt(e, ['id']))
        .where((e) => e > 0)
        .toList();

    final userIds = orders
        .map((e) => _readInt(e, ['created_by_user_id', 'user_id']))
        .where((e) => e > 0)
        .toSet()
        .toList();

    final Map<int, String> userNames = {};
    if (userIds.isNotEmpty) {
      final usersData = await supabase
          .from('users')
          .select('id, name')
          .inFilter('id', userIds);

      for (final row in (usersData as List)) {
        final map = row as Map<String, dynamic>;
        final id = _readInt(map, ['id']);
        final name = _readString(map, ['name'], fallback: 'Unknown');
        if (id > 0) userNames[id] = name;
      }
    }

    final itemsData = await supabase
        .from('order_items')
        .select('order_id, product_id, product_name, quantity, unit_price')
        .inFilter('order_id', orderIds);

    final itemRows = (itemsData as List).cast<Map<String, dynamic>>();

    final productIds = itemRows
        .map((e) => _readInt(e, ['product_id']))
        .where((e) => e > 0)
        .toSet()
        .toList();

    final Map<int, double> productPrices = {};
    if (productIds.isNotEmpty) {
      final productsData = await supabase
          .from('products')
          .select('id, price')
          .inFilter('id', productIds);

      for (final row in (productsData as List)) {
        final map = row as Map<String, dynamic>;
        final id = _readInt(map, ['id']);
        final price = _readDouble(map, ['price']);
        if (id > 0) productPrices[id] = price;
      }
    }

    return itemRows.map((map) {
      final orderId = _readInt(map, ['order_id']);
      final order = orders.firstWhere(
        (o) => _readInt(o, ['id']) == orderId,
        orElse: () => <String, dynamic>{},
      );

      final dt = DateTime.tryParse(
        (order['created_at'] ?? '').toString(),
      )?.toLocal();

      final quantity = _readInt(map, ['quantity', 'qty']);
      final productId = _readInt(map, ['product_id']);

      var unitPrice = _readDouble(map, ['unit_price']);
      if (unitPrice <= 0 && productId > 0) {
        unitPrice = productPrices[productId] ?? 0;
      }

      final userId = _readInt(order, ['created_by_user_id', 'user_id']);
      final soldBy = userNames[userId] ?? 'Unknown';

      return ReportSaleItem(
        productName: _readString(
          map,
          ['product_name'],
          fallback: 'Без име',
        ),
        quantity: quantity,
        unitPrice: unitPrice,
        totalEuro: quantity * unitPrice,
        soldBy: soldBy,
        time: dt,
      );
    }).toList();
  }

  Future<ClosedReceiptData?> getClosedReceiptByOrderId(int orderId) async {
    if (orderId <= 0) return null;

    final ordersData = await supabase
        .from('orders')
        .select('id, table_id, user_id, created_by_user_id, created_at')
        .eq('id', orderId)
        .limit(1);

    final orders = (ordersData as List).cast<Map<String, dynamic>>();
    if (orders.isEmpty) return null;

    final order = orders.first;
    final tableId = _readInt(order, ['table_id']);
    final userId = _readInt(order, ['created_by_user_id', 'user_id']);
    final receiptDate = DateTime.tryParse(
          _readString(order, ['created_at'], fallback: ''),
        )?.toLocal() ??
        DateTime.now();

    var tableName = tableId > 0 ? tableId.toString() : '-';
    var zoneName = '-';
    if (tableId > 0) {
      final tableData = await supabase
          .from('tables')
          .select('id, name, zone_id')
          .eq('id', tableId)
          .limit(1);
      final tableRows = (tableData as List).cast<Map<String, dynamic>>();
      if (tableRows.isNotEmpty) {
        final table = tableRows.first;
        tableName = _readString(table, ['name'], fallback: tableName);
        final zoneId = _readInt(table, ['zone_id']);
        if (zoneId > 0) {
          final zoneData = await supabase
              .from('zones')
              .select('name')
              .eq('id', zoneId)
              .limit(1);
          final zoneRows = (zoneData as List).cast<Map<String, dynamic>>();
          if (zoneRows.isNotEmpty) {
            zoneName = _readString(zoneRows.first, ['name'], fallback: '-');
          }
        }
      }
    }

    var waiterName = '-';
    if (userId > 0) {
      final userData = await supabase
          .from('users')
          .select('name')
          .eq('id', userId)
          .limit(1);
      final userRows = (userData as List).cast<Map<String, dynamic>>();
      if (userRows.isNotEmpty) {
        waiterName = _readString(userRows.first, ['name'], fallback: '-');
      }
    }

    final itemsData = await supabase
        .from('order_items')
        .select('product_name, quantity, unit_price')
        .eq('order_id', orderId);

    final items = (itemsData as List).cast<Map<String, dynamic>>().map((map) {
      final quantity = _readInt(map, ['quantity', 'qty']);
      final unitPrice = _readDouble(map, ['unit_price', 'price']);
      return ActiveOrderItem(
        productName: _readString(map, ['product_name', 'name'], fallback: 'Без име'),
        quantity: quantity <= 0 ? 1 : quantity,
        unitPrice: unitPrice,
        soldBy: waiterName,
        time: receiptDate,
      );
    }).toList();

    return ClosedReceiptData(
      orderId: orderId,
      tableName: tableName,
      zoneName: zoneName,
      waiterName: waiterName,
      items: items,
      receiptDate: receiptDate,
    );
  }
}

mixin RealtimeReloadMixin<T extends StatefulWidget> on State<T> {
  final List<RealtimeChannel> _channels = [];

  void watchTables(List<String> tables, VoidCallback onChange) {
    for (final table in tables) {
      final channel = supabase.channel(
        'watch-$table-${DateTime.now().microsecondsSinceEpoch}-${hashCode}',
      )
        ..onPostgresChanges(
          event: PostgresChangeEvent.all,
          schema: 'public',
          table: table,
          callback: (_) {
            if (mounted) onChange();
          },
        )
        ..subscribe();

      _channels.add(channel);
    }
  }

  @override
  void dispose() {
    for (final channel in _channels) {
      supabase.removeChannel(channel);
    }
    super.dispose();
  }
}

  Future<void> doLogin() async {
    final pin = pinController.text.trim();
    if (pin.isEmpty) {
      setState(() => error = 'Въведи PIN за $selectedRoleName');
      return;
    }

    setState(() {
      loading = true;
      error = '';
    });

    try {
      final user = await repo.tryUserLogin(pin, role: selectedRole);
      if (!mounted) return;

      if (user != null) {
        if (user.role == 1) {
          Navigator.pushReplacement(
            context,
            MaterialPageRoute(builder: (_) => AdminHomeScreen(admin: user)),
          );
        } else {
          Navigator.pushReplacement(
            context,
            MaterialPageRoute(builder: (_) => PosHomeScreen(posUser: user)),
          );
        }
      } else {
        setState(() => error = 'Грешен PIN или потребителят не е $selectedRoleName');
      }
    } catch (e) {
      setState(() => error = 'Грешка при връзка със Supabase: $e');
    } finally {
      if (mounted) setState(() => loading = false);
    }
  }

    super.initState();
    loadData();
    watchTables(['zones', 'tables', 'orders', 'order_items', 'users'], () {
      loadData(showLoader: false);
    });
  }

  Future<void> loadData({bool showLoader = true}) async {
    if (showLoader && mounted) setState(() => loading = true);
    try {
      zones = await repo.getZones();
      tables = await repo.getTables();
      users = await repo.getUsers();
      if (zones.isNotEmpty) {
        selectedZoneId ??= zones.first.id;
        if (!zones.any((z) => z.id == selectedZoneId)) selectedZoneId = zones.first.id;
      } else {
        selectedZoneId = null;
      }
    } catch (e) {
      if (mounted) showAppSnack(context, 'Грешка при зареждане на POS маси: $e', isError: true);

    final match = zones.where((e) => e.id == zoneId);
    return match.isEmpty ? '-' : match.first.name;

    if (userId == null || userId <= 0) return '-';
    final match = users.where((u) => u.id == userId);

    final busy = (table.status ?? 0) != 0 || (table.currentTotal ?? 0) > 0;

class PosOrderScreen extends StatefulWidget {
  final AppUserModel posUser;
  final TableModel table;
  final String zoneName;

  const PosOrderScreen({
    super.key,
    required this.posUser,
    required this.table,
    required this.zoneName,
  });

  @override
  State<PosOrderScreen> createState() => _PosOrderScreenState();
}

class _PosOrderScreenState extends State<PosOrderScreen> with RealtimeReloadMixin {
  bool loading = true;
  List<CategoryModel> categories = [];
  List<ProductModel> products = [];
  List<PosOrderLine> lockedItems = [];
  final List<PosOrderLine> draftItems = [];
  int? selectedCategoryId;
  bool _removingLockedItem = false;
  String? _selectedLockedItemKey;
  final Map<String, int> _removeQtyByKey = <String, int>{};

  String _lockedItemKey(PosOrderLine item) => '${item.productId}|${item.productName}|${item.unitPrice}';

  int removeQtyFor(PosOrderLine item) {
    final key = _lockedItemKey(item);
    final qty = _removeQtyByKey[key] ?? 1;
    return qty.clamp(1, item.quantity).toInt();
  }

  @override
  void initState() {
    super.initState();
    loadAll();
    watchTables(['orders', 'order_items', 'tables', 'products', 'categories'], () {
      loadAll(showLoader: false);
    });
  }

  Future<void> loadAll({bool showLoader = true}) async {
    if (showLoader && mounted) setState(() => loading = true);
    try {
      categories = (await repo.getCategories()).where((c) => c.isActive).toList();
      products = (await repo.getProducts()).where((p) => p.isActive).toList();
      lockedItems = await repo.getOpenOrderLinesForTable(widget.table.id, userId: widget.posUser.id);
      if (_selectedLockedItemKey != null &&
          !lockedItems.any((x) => _lockedItemKey(x) == _selectedLockedItemKey)) {
        _selectedLockedItemKey = null;
      }
      for (final item in lockedItems) {
        final key = _lockedItemKey(item);
        final oldQty = _removeQtyByKey[key] ?? 1;
        _removeQtyByKey[key] = oldQty.clamp(1, item.quantity).toInt();
      }
      _removeQtyByKey.removeWhere((key, value) => !lockedItems.any((x) => _lockedItemKey(x) == key));
      selectedCategoryId ??= categories.isEmpty ? null : categories.first.id;
      if (selectedCategoryId != null && !categories.any((c) => c.id == selectedCategoryId)) {
        selectedCategoryId = categories.isEmpty ? null : categories.first.id;
      }
    } catch (e) {
      if (mounted) showAppSnack(context, e.toString(), isError: true);
    }
    if (mounted) setState(() => loading = false);
  }

  List<ProductModel> get visibleProducts {
    if (selectedCategoryId == null) return products;
    return products.where((p) => p.categoryId == selectedCategoryId).toList();
  }

  double get lockedTotal => lockedItems.fold<double>(0, (sum, item) => sum + item.total);
  double get draftTotal => draftItems.fold<double>(0, (sum, item) => sum + item.total);
  double get total => lockedTotal + draftTotal;

  void addProductToDraft(ProductModel product) {
    final index = draftItems.indexWhere((x) => x.productId == product.id && x.unitPrice == product.price);
    setState(() {
      if (index >= 0) {
        final old = draftItems[index];
        draftItems[index] = PosOrderLine(
          productId: old.productId,
          productName: old.productName,
          unitPrice: old.unitPrice,
          quantity: old.quantity + 1,
        );
      } else {
        draftItems.add(PosOrderLine(
          productId: product.id,
          productName: product.name,
          quantity: 1,
          unitPrice: product.price,
        ));
      }
    });
  }

  void changeDraftQty(PosOrderLine item, int delta) {
    final index = draftItems.indexOf(item);
    if (index < 0) return;
    setState(() {
      final newQty = item.quantity + delta;
      if (newQty <= 0) {
        draftItems.removeAt(index);
      } else {
        draftItems[index] = PosOrderLine(
          productId: item.productId,
          productName: item.productName,
          unitPrice: item.unitPrice,
          quantity: newQty,
        );
      }
    });
  }


  void selectLockedItemLikeWpf(PosOrderLine item) {
    final key = _lockedItemKey(item);
    setState(() {
      if (_selectedLockedItemKey == key) {
        _selectedLockedItemKey = null;
      } else {
        _selectedLockedItemKey = key;
        _removeQtyByKey[key] = (_removeQtyByKey[key] ?? 1).clamp(1, item.quantity).toInt();
      }
    });
  }

  void changeLockedRemoveQty(PosOrderLine item, int delta) {
    final key = _lockedItemKey(item);
    setState(() {
      _selectedLockedItemKey = key;
      final current = _removeQtyByKey[key] ?? 1;
      _removeQtyByKey[key] = (current + delta).clamp(1, item.quantity).toInt();
    });
  }

  Future<void> removeSelectedLockedItemLikeWpf(PosOrderLine item) async {
    final key = _lockedItemKey(item);
    if (_selectedLockedItemKey != key) {
      showAppSnack(
        context,
        'Първо маркирай продукта, както в WPF.',
        type: AppNoticeType.warning,
      );
      return;
    }
    await removeLockedItem(item, qtyToRemove: removeQtyFor(item));
  }

  Future<void> placeDraftOrder() async {
    if (draftItems.isEmpty) {
      showAppSnack(context, 'Няма добавени продукти за поръчване', isError: true);
      return;
    }

    try {
      await repo.placeOrderItems(
        tableId: widget.table.id,
        userId: widget.posUser.id,
        items: List<PosOrderLine>.from(draftItems),
      );
      draftItems.clear();
      await loadAll(showLoader: false);
      if (mounted) showAppSnack(context, 'Поръчката е изпратена');
    } catch (e) {
      if (mounted) showAppSnack(context, e.toString(), isError: true);
    }
  }

  Future<bool> askAdminPin() async {
    final ok = await Navigator.of(context).push<bool>(
      MaterialPageRoute(
        fullscreenDialog: true,
        builder: (_) => const AdminPinConfirmScreen(),
      ),
    );

    return ok == true;
  }

  Future<void> removeLockedItem(PosOrderLine item, {required int qtyToRemove}) async {
    if (_removingLockedItem) return;

    final safeQty = qtyToRemove.clamp(1, item.quantity).toInt();
    setState(() => _removingLockedItem = true);

    try {
      final ok = await askAdminPin();
      if (!mounted || !ok) return;

      // WPF начин: маркираш продукта, избираш колко броя да махнеш, натискаш минус,
      // въвеждаш admin PIN и директно се махат избраните бройки без втори dialog.
      await repo.placeOrderItems(
        tableId: widget.table.id,
        userId: widget.posUser.id,
        items: [
          PosOrderLine(
            productId: item.productId,
            productName: item.productName,
            quantity: -safeQty,
            unitPrice: item.unitPrice,
          ),
        ],
      );

      if (!mounted) return;
      final key = _lockedItemKey(item);
      setState(() {
        _selectedLockedItemKey = null;
        _removeQtyByKey.remove(key);
      });
      await loadAll(showLoader: false);

      if (mounted) {
        showAppSnack(
          context,
          'Премахнати са $safeQty бр. от ${item.productName}',
          type: AppNoticeType.success,
        );
      }
    } catch (e) {
      if (mounted) {
        showAppSnack(context, e.toString(), isError: true);
      }
    } finally {
      if (mounted) setState(() => _removingLockedItem = false);
    }
  }

  List<ActiveOrderItem> _receiptItemsFromPosLines(List<PosOrderLine> lines) {
    return lines
        .where((x) => x.quantity > 0)
        .map(
          (x) => ActiveOrderItem(
            productName: x.productName,
            quantity: x.quantity,
            unitPrice: x.unitPrice,
            soldBy: widget.posUser.name,
            time: DateTime.now(),
          ),
        )
        .toList();
  }

  Future<void> savePosReceiptPng({
    required List<PosOrderLine> receiptLines,
    required double receiptTotal,
    required String paymentMethod,
    double? originalTotal,
    double discountPercent = 0,
    double discountAmount = 0,
  }) async {
    final printableItems = _receiptItemsFromPosLines(receiptLines);
    if (printableItems.isEmpty) {
      showAppSnack(context, 'Няма продукти за печат', isError: true);
      return;
    }

    try {
      final now = DateTime.now();
      final imageBytes = await ScreenshotController().captureFromWidget(
        ReceiptImageView(
          tableName: widget.table.name,
          zoneName: widget.zoneName,
          waiterName: widget.posUser.name,
          items: printableItems,
          total: receiptTotal,
          originalTotal: originalTotal,
          discountPercent: discountPercent,
          discountAmount: discountAmount,
          paymentMethod: paymentMethod,
          receiptDate: now,
        ),
        context: context,
        pixelRatio: 3,
        targetSize: const Size(384, 900),
      );

      final dir = await getTemporaryDirectory();
      final file = File(
        '${dir.path}/flexi_pos_receipt_${widget.table.name}_${now.millisecondsSinceEpoch}.png',
      );
      await file.writeAsBytes(imageBytes);

      final hasAccess = await Gal.hasAccess();
      if (!hasAccess) {
        await Gal.requestAccess();
      }

      await Gal.putImage(file.path, album: 'FLEXI Receipts');

      if (mounted) {
        showAppSnack(context, 'Бележката е записана в Галерията / Google Photos');
      }
    } catch (e) {
      if (mounted) {
        showAppSnack(context, 'Грешка при печат/запис на бележка: $e', isError: true);
      }
    }
  }

  Future<void> closeBill({
    required String paymentMethod,
    required bool useEsp32,
    required bool printReceipt,
  }) async {
    if (lockedItems.isEmpty && draftItems.isEmpty) {
      showAppSnack(context, 'Няма продукти за приключване', isError: true);
      return;
    }

    if (draftItems.isNotEmpty) {
      await placeDraftOrder();
      await loadAll(showLoader: false);
    }

    final receiptLines = List<PosOrderLine>.from(lockedItems);
    final receiptTotal = receiptLines.fold<double>(0, (sum, item) => sum + item.total);
    double finalReceiptTotal = receiptTotal;
    double discountPercent = 0;
    double discountAmount = 0;
    String finalPaymentMethod = paymentMethod;

    final confirm = await showDialog<bool>(
      context: context,
      builder: (_) => AlertDialog(
        backgroundColor: panelColor,
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(24),
          side: const BorderSide(color: borderBlue, width: 1.5),
        ),
        title: const Text('Приключване на сметка'),
        content: Text(
          'Маса ${widget.table.name}\nПлащане: $paymentMethod\nОбщо: ${euro(receiptTotal)}\n\n${useEsp32 ? 'Ще се изчака ESP32 плащане и после ще се запази бележка.' : 'Ще се запази бележка без ESP32 плащане.'}',
          style: const TextStyle(color: whiteText, height: 1.35),
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(context, false), child: const Text('Отказ')),
          ElevatedButton(onPressed: () => Navigator.pop(context, true), child: const Text('Приключи')),
        ],
      ),
    );

    if (confirm != true) return;

    try {
      if (useEsp32) {
        showDialog(
          context: context,
          barrierDismissible: false,
          builder: (_) {
            return AlertDialog(
              backgroundColor: panelColor,
              shape: RoundedRectangleBorder(
                borderRadius: BorderRadius.circular(24),
                side: const BorderSide(color: borderPurple, width: 1.5),
              ),
              title: const Text('ESP32 плащане'),
              content: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Text(
                    'Сума: ${euro(receiptTotal)}',
                    style: const TextStyle(color: greenText, fontSize: 24, fontWeight: FontWeight.bold),
                  ),
                  const SizedBox(height: 12),
                  const Text(
                    'Доближи картата до ESP32 и въведи PIN.',
                    textAlign: TextAlign.center,
                    style: TextStyle(color: whiteText),
                  ),
                  const SizedBox(height: 18),
                  const CircularProgressIndicator(),
                ],
              ),
              actions: [
                TextButton(
                  onPressed: () async {
                    await cancelEsp32Payment();
                    if (context.mounted) Navigator.pop(context);
                  },
                  child: const Text('Отказ'),
                ),
              ],
            );
          },
        );

        final started = await startEsp32Payment(
          amount: receiptTotal,
          orderId: widget.table.id,
          tableName: widget.table.name,
        );

        if (!mounted) return;

        if (!started) {
          Navigator.pop(context);
          showAppSnack(context, 'ESP32 не отговори. Провери IP-то и WiFi връзката.', isError: true);
          return;
        }

        final paymentResult = await waitForEsp32PaymentResult();

        if (!mounted) return;
        Navigator.pop(context);

        if (!paymentResult.approved) {
          showAppSnack(context, 'Плащането е отказано или изтече времето.', isError: true);
          return;
        }

        discountPercent = paymentResult.discountPercent;
        discountAmount = paymentResult.discountAmount;
        finalReceiptTotal = paymentResult.finalAmount > 0
            ? paymentResult.finalAmount
            : receiptTotal - discountAmount;

        if (paymentResult.hasFlexiDiscount) {
          finalPaymentMethod = 'Карта / ESP32';
          showAppSnack(
            context,
            'Flexi карта разпозната: приложена е 5% отстъпка',
          );
        }
      }

      if (printReceipt) {
        await savePosReceiptPng(
          receiptLines: receiptLines,
          receiptTotal: finalReceiptTotal,
          originalTotal: receiptTotal,
          discountPercent: discountPercent,
          discountAmount: discountAmount,
          paymentMethod: finalPaymentMethod,
        );
      }

      await repo.closeOrder(
        tableId: widget.table.id,
        userId: widget.posUser.id,
        paymentMethod: finalPaymentMethod,
        discountPercent: discountPercent,
      );

      if (mounted) {
        showAppSnack(context, useEsp32 ? 'ESP32 плащането е успешно и сметката е приключена' : 'Бележката е записана и сметката е приключена');
        Navigator.pop(context);
      }
    } catch (e) {
      if (mounted) showAppSnack(context, e.toString(), isError: true);
    }
  }

                          final newPin = pinController.text.trim();
                          if (newPin.isEmpty) {
                            throw Exception('PIN не може да е празен');
                          }
                          if (newPin != user.pinCode) {
                            await repo.updateUserPin(user.id, newPin);

class _TablesScreenState extends State<TablesScreen> with RealtimeReloadMixin {
  bool loading = true;
  List<ZoneModel> zones = [];
  List<TableModel> tables = [];
  int? selectedZoneId;

  @override
  void initState() {
    super.initState();
    loadData();
    watchTables(['zones', 'tables', 'orders', 'order_items'], () {
      loadData(showLoader: false);
    });
  }

  Future<void> loadData({bool showLoader = true}) async {
    if (showLoader && mounted) setState(() => loading = true);
    try {
      zones = await repo.getZones();
      tables = await repo.getTables();

      if (zones.isNotEmpty) {
        selectedZoneId ??= zones.first.id;
        if (!zones.any((z) => z.id == selectedZoneId)) {
          selectedZoneId = zones.first.id;
        }
      } else {
        selectedZoneId = null;
      }
    } catch (e) {
      if (mounted) showAppSnack(context, 'Грешка при зареждане на маси/зони: $e', isError: true);
    }
    if (mounted) setState(() => loading = false);
  }

  List<TableModel> get filteredTables {
    if (selectedZoneId == null) return [];
    return tables.where((e) => e.zoneId == selectedZoneId).toList();
  }

  String zoneName(int zoneId) {
    final match = zones.where((e) => e.id == zoneId);
    return match.isEmpty ? '-' : match.first.name;
  }

    final controller = TextEditingController();

class _TableDetailsScreenState extends State<TableDetailsScreen> with RealtimeReloadMixin {
  bool loading = true;
  List<ActiveOrderItem> items = [];
  bool _firstLoadDone = false;
  bool _closedReceiptDialogOpen = false;

  @override
  void initState() {
    super.initState();
    loadItems();
    watchTables(['orders', 'order_items', 'users', 'tables'], () {
      loadItems(showLoader: false);
    });
  }

  Future<void> loadItems({bool showLoader = true}) async {
    final previousItems = List<ActiveOrderItem>.from(items);
    final previousTotal = previousItems.fold<double>(0, (sum, item) => sum + item.total);

    if (showLoader && mounted) setState(() => loading = true);

    try {
      final newItems = await repo.getActiveItemsForTable(widget.table.id);

      final shouldShowClosedReceipt =
          _firstLoadDone &&
          previousItems.isNotEmpty &&
          newItems.isEmpty &&
          !_closedReceiptDialogOpen;

      if (mounted) {
        setState(() {
          items = newItems;
          loading = false;
          _firstLoadDone = true;
        });
      }

      if (shouldShowClosedReceipt && mounted) {
        WidgetsBinding.instance.addPostFrameCallback((_) {
          if (!mounted) return;
          showClosedReceiptDialog(previousItems, previousTotal);
        });
      }
    } catch (e) {
      if (mounted) {
        setState(() {
          items = [];
          loading = false;
          _firstLoadDone = true;
        });
        showAppSnack(context, 'Грешка при зареждане на поръчки: $e', isError: true);
      }
    }
  }

  double get total {
    double sum = 0;
    for (final item in items) {
      sum += item.total;
    }
    return sum;
  }

  Future<void> finishWithEsp32Payment() async {
    if (items.isEmpty) {
      showAppSnack(context, 'Няма продукти за плащане', isError: true);
      return;
    }

    final paymentTotal = total;
    final receiptItems = List<ActiveOrderItem>.from(items);
    final receiptDate = DateTime.now();

    showDialog(
      context: context,
      barrierDismissible: false,
      builder: (_) {
        return AlertDialog(
          backgroundColor: panelColor,
          title: const Text('Плащане с карта'),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text(
                'Сума: ${euro(paymentTotal)}',
                style: const TextStyle(
                  color: greenText,
                  fontSize: 22,
                  fontWeight: FontWeight.bold,
                ),
              ),
              const SizedBox(height: 12),
              const Text(
                'Доближи RFID картата до ESP32 и въведи PIN.\nПосле натисни # на клавиатурата.',
                style: TextStyle(color: whiteText),
              ),
              const SizedBox(height: 18),
              const Center(child: CircularProgressIndicator()),
            ],
          ),
          actions: [
            TextButton(
              onPressed: () async {
                await cancelEsp32Payment();
                if (context.mounted) Navigator.pop(context);
              },
              child: const Text('Отказ'),
            ),
          ],
        );
      },
    );

    final started = await startEsp32Payment(
      amount: paymentTotal,
      orderId: widget.table.id,
      tableName: widget.table.name,
    );

    if (!mounted) return;

    if (!started) {
      Navigator.pop(context);
      showAppSnack(
        context,
        'ESP32 не отговори. Провери IP-то и WiFi връзката.',
        isError: true,
      );
      return;
    }

    final approved = await waitForEsp32Payment();

    if (!mounted) return;
    Navigator.pop(context);

    if (!approved) {
      showAppSnack(context, 'Плащането е отказано или изтече времето.', isError: true);
      return;
    }

    showAppSnack(context, 'Плащането е успешно');

    await saveReceiptPngToGooglePhotos(
      receiptItems: receiptItems,
      receiptTotal: paymentTotal,
      receiptDate: receiptDate,
    );
  }

  Future<void> saveReceiptPngToGooglePhotos({
    List<ActiveOrderItem>? receiptItems,
    double? receiptTotal,
    DateTime? receiptDate,
    double? originalTotal,
    double discountPercent = 0,
    double discountAmount = 0,
  }) async {
    final printableItems = receiptItems ?? items;
    final printableTotal = receiptTotal ??
        printableItems.fold<double>(0, (sum, item) => sum + item.total);
    final date = receiptDate ?? DateTime.now();

    if (printableItems.isEmpty) {
      showAppSnack(context, 'Няма продукти за печат', isError: true);
      return;
    }

    try {
      final imageBytes = await ScreenshotController().captureFromWidget(
        ReceiptImageView(
          tableName: widget.table.name,
          zoneName: widget.zoneName,
          waiterName: printableItems.isEmpty ? '-' : printableItems.first.soldBy,
          items: printableItems,
          total: printableTotal,
          originalTotal: originalTotal,
          discountPercent: discountPercent,
          discountAmount: discountAmount,
          receiptDate: date,
        ),
        context: context,
        pixelRatio: 3,
        targetSize: const Size(384, 900),
      );

      final dir = await getTemporaryDirectory();
      final fileName =
          'flexi_receipt_${widget.table.name}_${DateTime.now().millisecondsSinceEpoch}.png';
      final file = File('${dir.path}/$fileName');
      await file.writeAsBytes(imageBytes);

      final hasAccess = await Gal.hasAccess();
      if (!hasAccess) {
        await Gal.requestAccess();
      }

      await Gal.putImage(file.path, album: 'FLEXI Receipts');

      if (mounted) {
        showAppSnack(
          context,
          'PNG бележката е записана в Галерията / Google Photos',
        );
      }
    } catch (e) {
      if (mounted) {
        showAppSnack(context, 'Грешка при запис на PNG бележка: $e', isError: true);
      }
    }
  }

  Future<void> finishClosedReceiptWithEsp32({
    required BuildContext dialogContext,
    required List<ActiveOrderItem> receiptItems,
    required double receiptTotal,
    required DateTime receiptDate,
  }) async {
    try {
      final started = await startEsp32Payment(
        amount: receiptTotal,
        orderId: widget.table.id,
        tableName: widget.table.name,
      );

      if (!mounted) return;

      if (!started) {
        showAppSnack(context, 'ESP32 не отговори. Провери IP-то и WiFi връзката.', isError: true);
        return;
      }

      showAppSnack(context, 'Очаква се карта на ESP32...', type: AppNoticeType.info);

      final paymentResult = await waitForEsp32PaymentResult();
      if (!mounted) return;

      if (!paymentResult.approved) {
        showAppSnack(context, 'Плащането е отказано или изтече времето.', isError: true);
        return;
      }

      final discountPercent = paymentResult.discountPercent;
      final discountAmount = paymentResult.discountAmount;
      final finalTotal = paymentResult.finalAmount > 0
          ? paymentResult.finalAmount
          : receiptTotal - discountAmount;

      await saveReceiptPngToGooglePhotos(
        receiptItems: receiptItems,
        receiptTotal: finalTotal,
        originalTotal: receiptTotal,
        discountPercent: discountPercent,
        discountAmount: discountAmount,
        receiptDate: receiptDate,
      );

      if (dialogContext.mounted) Navigator.pop(dialogContext);
      if (mounted) {
        showAppSnack(
          context,
          paymentResult.hasFlexiDiscount
              ? 'ESP32 платено успешно • Flexi карта - 5%'
              : 'ESP32 платено успешно',
          type: AppNoticeType.success,
        );
      }
    } catch (e) {
      if (mounted) showAppSnack(context, 'Грешка при ESP32 плащане: $e', isError: true);
    }
  }

class GlobalClosedReceiptWatcher {
  final GlobalKey<NavigatorState> navigatorKey;
  RealtimeChannel? _channel;
  bool _dialogOpen = false;
  final Set<int> _shownOrderIds = <int>{};

  GlobalClosedReceiptWatcher({required this.navigatorKey});

  void start() {
    _channel = supabase.channel('global-closed-orders-watch')
      ..onPostgresChanges(
        event: PostgresChangeEvent.update,
        schema: 'public',
        table: 'orders',
        callback: (payload) async {
          final record = payload.newRecord;
          if (!_isClosedStatus(record['status'])) return;

          final orderId = _readOrderId(record);
          if (orderId <= 0 || _shownOrderIds.contains(orderId)) return;
          _shownOrderIds.add(orderId);

          final context = navigatorKey.currentContext;
          if (context == null) return;

          ClosedReceiptData? receipt;
          try {
            receipt = await repo.getClosedReceiptByOrderId(orderId);
          } catch (_) {
            receipt = null;
          }

          final latestContext = navigatorKey.currentContext;
          if (latestContext == null) return;
          await _showClosedDialog(latestContext, receipt, orderId);
        },
      )
      ..subscribe();
  }

  void stop() {
    final channel = _channel;
    if (channel != null) {
      supabase.removeChannel(channel);
    }
    _channel = null;
  }

  int _readOrderId(Map<String, dynamic> record) {
    final value = record['id'];
    if (value is int) return value;
    if (value is num) return value.toInt();
    return int.tryParse(value?.toString() ?? '') ?? 0;
  }

  bool _isClosedStatus(dynamic status) {
    if (status == null) return false;
    if (status is num) return status.toInt() == 2;
    final text = status.toString().trim().toLowerCase();
    return text == 'closed' ||
        text == 'paid' ||
        text == 'finished' ||
        text == 'completed' ||
        text == 'приключена' ||
        text == 'платена';
  }

  Future<void> _showClosedDialog(
    BuildContext context,
    ClosedReceiptData? receipt,
    int orderId,
  ) async {
    if (_dialogOpen) return;
    _dialogOpen = true;

    try {
      await showDialog(
        context: context,
        barrierDismissible: false,
        builder: (dialogContext) {
          final hasItems = receipt != null && receipt.items.isNotEmpty;
          final tableText = receipt == null ? '-' : receipt.tableName;
          final totalText = receipt == null ? '-' : euro(receipt.total);
          bool esp32Busy = false;

          return StatefulBuilder(
            builder: (dialogContext, setDialogState) {
              return AlertDialog(
                backgroundColor: panelColor,
                title: const Text('Сметката е приключена'),
                content: Text(
                  'Маса: $tableText\nОбщо: $totalText\n\nИзбери печат или ESP32 + печат.',
                  style: const TextStyle(color: whiteText),
                ),
                actions: [
                  TextButton(
                    onPressed: esp32Busy ? null : () => Navigator.pop(dialogContext),
                    child: const Text('Затвори'),
                  ),
                  ElevatedButton.icon(
                    icon: const Icon(Icons.print_outlined),
                    label: const Text('Печат'),
                    onPressed: hasItems && !esp32Busy
                        ? () async {
                            await _saveReceiptToGooglePhotos(dialogContext, receipt!);
                            if (dialogContext.mounted) Navigator.pop(dialogContext);
                          }
                        : null,
                  ),
                  ElevatedButton.icon(
                    icon: esp32Busy
                        ? const SizedBox(
                            width: 18,
                            height: 18,
                            child: CircularProgressIndicator(strokeWidth: 2),
                          )
                        : const Icon(Icons.credit_card),
                    label: Text(esp32Busy ? 'Изчакване...' : 'ESP32 + печат'),
                    onPressed: hasItems && !esp32Busy
                        ? () async {
                            setDialogState(() => esp32Busy = true);
                            await _payClosedReceiptWithEsp32AndPrint(dialogContext, receipt!);
                            if (dialogContext.mounted) {
                              setDialogState(() => esp32Busy = false);
                            }
                          }
                        : null,
                  ),
                ],
              );
            },
          );
        },
      );
    } finally {
      _dialogOpen = false;
    }
  }

  Future<void> _saveReceiptToGooglePhotos(
    BuildContext context,
    ClosedReceiptData receipt, {
    double? total,
    double? originalTotal,
    double discountPercent = 0,
    double discountAmount = 0,
  }) async {
    try {
      final imageBytes = await ScreenshotController().captureFromWidget(
        ReceiptImageView(
          tableName: receipt.tableName,
          zoneName: receipt.zoneName,
          waiterName: receipt.waiterName,
          items: receipt.items,
          total: total ?? receipt.total,
          originalTotal: originalTotal,
          discountPercent: discountPercent,
          discountAmount: discountAmount,
          receiptDate: receipt.receiptDate,
        ),
        context: context,
        pixelRatio: 3,
        targetSize: const Size(384, 900),
      );

      final dir = await getTemporaryDirectory();
      final file = File(
        '${dir.path}/flexi_receipt_${receipt.orderId}_${DateTime.now().millisecondsSinceEpoch}.png',
      );
      await file.writeAsBytes(imageBytes);

      final hasAccess = await Gal.hasAccess();
      if (!hasAccess) {
        await Gal.requestAccess();
      }

      await Gal.putImage(file.path, album: 'FLEXI Receipts');

      if (context.mounted) {
        showAppSnack(
          context,
          'PNG бележката е записана в Галерията / Google Photos',
          type: AppNoticeType.success,
        );
      }
    } catch (e) {
      if (context.mounted) {
        showAppSnack(context, 'Грешка при запис на PNG бележка: $e', isError: true);
      }
    }
  }

  Future<void> _payClosedReceiptWithEsp32AndPrint(
    BuildContext context,
    ClosedReceiptData receipt,
  ) async {
    try {
      final started = await startEsp32Payment(
        amount: receipt.total,
        orderId: receipt.orderId,
        tableName: receipt.tableName,
      );

      if (!context.mounted) return;

      if (!started) {
        showAppSnack(context, 'ESP32 не отговори. Провери IP-то и WiFi връзката.', isError: true);
        return;
      }

      showAppSnack(context, 'Очаква се карта на ESP32...', type: AppNoticeType.info);

      final paymentResult = await waitForEsp32PaymentResult();

      if (!context.mounted) return;

      if (!paymentResult.approved) {
        showAppSnack(context, 'Плащането е отказано или изтече времето.', isError: true);
        return;
      }

      final discountPercent = paymentResult.discountPercent;
      final discountAmount = paymentResult.discountAmount;
      final finalTotal = paymentResult.finalAmount > 0
          ? paymentResult.finalAmount
          : receipt.total - discountAmount;

      await _saveReceiptToGooglePhotos(
        context,
        receipt,
        total: finalTotal,
        originalTotal: receipt.total,
        discountPercent: discountPercent,
        discountAmount: discountAmount,
      );

      if (context.mounted) Navigator.pop(context);

      final latestContext = navigatorKey.currentContext;
      if (latestContext != null) {
        showAppSnack(
          latestContext,
          paymentResult.hasFlexiDiscount
              ? 'ESP32 платено успешно • Flexi карта - 5%'
              : 'ESP32 платено успешно',
          type: AppNoticeType.success,
        );
      }
    } catch (e) {
      if (context.mounted) {
        showAppSnack(context, 'Грешка при ESP32 плащане: $e', isError: true);
      }
    }
  }

}

  late int selectedYear;
  late int selectedMonth;
  int? selectedDay;
  bool loading = false;
  List<ReportSaleItem> items = [];

  final List<int> years = List.generate(7, (index) => 2024 + index);

  final List<String> monthNames = const [
    'Януари',
    'Февруари',
    'Март',
    'Април',
    'Май',
    'Юни',
    'Юли',
    'Август',
    'Септември',
    'Октомври',
    'Ноември',
    'Декември',
  ];

  final List<String> weekDays = const ['ПН', 'ВТ', 'СР', 'ЧТ', 'ПТ', 'СБ', 'НД'];

  @override
  void initState() {
    super.initState();
    final now = DateTime.now();
    selectedYear = now.year;
    selectedMonth = now.month;
    selectedDay = now.day;
    loadReport();
    watchTables(['orders', 'order_items', 'users'], () {
      loadReport(showLoader: false);
    });
  }

  Future<void> loadReport({bool showLoader = true}) async {
    if (selectedDay == null) return;
    if (showLoader && mounted) setState(() => loading = true);
    try {
      items = await repo.getDailyReport(DateTime(selectedYear, selectedMonth, selectedDay!));
    } catch (e) {
      items = [];
      if (mounted) showAppSnack(context, 'Грешка при отчета: $e', isError: true);
    }
    if (mounted) setState(() => loading = false);
  }

  int get totalQty => items.fold(0, (a, b) => a + b.quantity);
  double get totalAmount => items.fold(0, (a, b) => a + b.totalEuro);

  int daysInMonth(int year, int month) => DateTime(year, month + 1, 0).day;
  int firstWeekdayOffset(int year, int month) => DateTime(year, month, 1).weekday - 1;

  Widget buildCalendarGrid() {
    final totalDays = daysInMonth(selectedYear, selectedMonth);
    final offset = firstWeekdayOffset(selectedYear, selectedMonth);
    final totalCells = ((offset + totalDays) / 7).ceil() * 7;

    return GridView.builder(
      shrinkWrap: true,
      physics: const NeverScrollableScrollPhysics(),
      itemCount: totalCells,
      gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
        crossAxisCount: 7,
        mainAxisSpacing: 8,
        crossAxisSpacing: 8,
        childAspectRatio: 1.0,
      ),
      itemBuilder: (context, index) {
        final dayNumber = index - offset + 1;
        if (index < offset || dayNumber > totalDays) {
          return const SizedBox.shrink();
        }

        final isSelected = selectedDay == dayNumber;

        return GestureDetector(
          onTap: () async {
            setState(() => selectedDay = dayNumber);
            await loadReport();
          },
          child: AnimatedContainer(
            duration: const Duration(milliseconds: 180),
            decoration: BoxDecoration(
              color: isSelected ? const Color(0xFF0A3D57) : panelInner,
              borderRadius: BorderRadius.circular(12),
              border: Border.all(
                color: isSelected ? borderBlue : const Color(0xFF22336D),
                width: isSelected ? 2 : 1.2,
              ),
            ),
            child: Center(
              child: Text(
                '$dayNumber',
                style: const TextStyle(
                  color: whiteText,
                  fontWeight: FontWeight.w700,
                ),
              ),
            ),
          ),
        );
      },
    );
  }

  Widget buildCalendarPanel() {
    return AppThemeBox(
      borderColor: borderBlue,
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const SectionTitle('Календар', subtitle: 'Избери месец, година и ден'),
          const SizedBox(height: 14),
          DropdownButtonFormField<int>(
            value: selectedMonth,
            dropdownColor: panelInner,
            decoration: const InputDecoration(
              filled: true,
              fillColor: Color(0xFF0C1538),
              border: OutlineInputBorder(),
            ),
            items: List.generate(12, (index) {
              final monthNumber = index + 1;
              return DropdownMenuItem<int>(
                value: monthNumber,
                child: Text(monthNames[index]),
              );
            }),
            onChanged: (value) async {
              if (value == null) return;
              final maxDay = daysInMonth(selectedYear, value);
              setState(() {
                selectedMonth = value;
                selectedDay = selectedDay == null ? 1 : selectedDay!.clamp(1, maxDay);
              });
              await loadReport();
            },
          ),
          const SizedBox(height: 12),
          DropdownButtonFormField<int>(
            value: selectedYear,
            dropdownColor: panelInner,
            decoration: const InputDecoration(
              filled: true,
              fillColor: Color(0xFF0C1538),
              border: OutlineInputBorder(),
            ),
            items: years
                .map(
                  (year) => DropdownMenuItem<int>(
                    value: year,
                    child: Text(year.toString()),
                  ),
                )
                .toList(),
            onChanged: (value) async {
              if (value == null) return;
              final maxDay = daysInMonth(value, selectedMonth);
              setState(() {
                selectedYear = value;
                selectedDay = selectedDay == null ? 1 : selectedDay!.clamp(1, maxDay);
              });
              await loadReport();
            },
          ),
          const SizedBox(height: 16),
          Row(
            children: weekDays
                .map(
                  (day) => Expanded(
                    child: Center(
                      child: Padding(
                        padding: const EdgeInsets.only(bottom: 10),
                        child: Text(
                          day,
                          style: const TextStyle(
                            color: mutedText,
                            fontSize: 13,
                            fontWeight: FontWeight.w700,
                          ),
                        ),
                      ),
                    ),
                  ),
                )
                .toList(),
          ),
          buildCalendarGrid(),
        ],
      ),
    );
  }
