#include <WiFi.h>
#include <WebServer.h>
#include <SPI.h>
#include <MFRC522.h>
#include <Keypad.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <ArduinoJson.h>

#define SS_PIN 5
#define RST_PIN 4

MFRC522 rfid(SS_PIN, RST_PIN);
LiquidCrystal_I2C lcd(0x27, 16, 2);
WebServer server(80);

const char* ssid = "Flexi Bar";
const char* password = "flexi1234";

String paymentStatus = "idle";
float paymentAmount = 0.0;
float finalAmount = 0.0;
float discountAmount = 0.0;
float discountPercent = 0.0;

String enteredPin = "";
String currentCardName = "";
String currentCardUid = "";

// CARD 1: Flexi - 5% discount
String flexiUid = "A1514166";
String flexiPin = "431";

// CARD 2: Nexus - no discount
String nexusUid = "B046FC55";
String nexusPin = "676";

// CARD 3: Blaze - no discount
String blazeUid = "A07FFE55";
String blazePin = "831";

bool waitingForCard = false;
bool waitingForPin = false;

char keys[4][3] = {
  {'1','2','3'},
  {'4','5','6'},
  {'7','8','9'},
  {'*','0','#'}
};

byte rowPins[4] = {13, 12, 14, 27};
byte colPins[3] = {26, 25, 33};

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, 4, 3);

void showReady() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("FLEXI POS");
  lcd.setCursor(0, 1);
  lcd.print("READY");
}

void resetCardData() {
  enteredPin = "";
  currentCardName = "";
  currentCardUid = "";
  discountPercent = 0.0;
  discountAmount = 0.0;
  finalAmount = paymentAmount;
}

void showScanCard() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("SUMA:");
  lcd.print(paymentAmount, 2);

  lcd.setCursor(0, 1);
  lcd.print("SCAN CARD");
}

void handleStartPayment() {
  if (server.method() != HTTP_POST) {
    server.send(405, "text/plain", "Only POST");
    return;
  }

  DynamicJsonDocument doc(256);
  DeserializationError error = deserializeJson(doc, server.arg("plain"));

  if (error) {
    server.send(400, "application/json", "{\"success\":false,\"message\":\"invalid_json\"}");
    return;
  }

  paymentAmount = doc["amount"] | 0.0;
  finalAmount = paymentAmount;

  paymentStatus = "waiting_card";
  waitingForCard = true;
  waitingForPin = false;
  resetCardData();

  showScanCard();

  server.send(200, "application/json", "{\"success\":true}");
}

void handlePaymentStatus() {
  DynamicJsonDocument doc(256);

  doc["status"] = paymentStatus;
  doc["card"] = currentCardName;
  doc["uid"] = currentCardUid;
  doc["discountPercent"] = discountPercent;
  doc["discountAmount"] = discountAmount;
  doc["amount"] = paymentAmount;
  doc["finalAmount"] = finalAmount;

  String response;
  serializeJson(doc, response);

  server.send(200, "application/json", response);
}

void handleCancelPayment() {
  paymentStatus = "cancelled";
  waitingForCard = false;
  waitingForPin = false;
  resetCardData();

  showReady();

  server.send(200, "application/json", "{\"success\":true}");
}

String readCardUid() {
  String uid = "";

  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) {
      uid += "0";
    }

    uid += String(rfid.uid.uidByte[i], HEX);
  }

  uid.toUpperCase();
  return uid;
}

bool selectCardByUid(String uid) {
  if (uid == flexiUid) {
    currentCardName = "flexi";
    currentCardUid = uid;
    discountPercent = 5.0;
    discountAmount = paymentAmount * discountPercent / 100.0;
    finalAmount = paymentAmount - discountAmount;
    return true;
  }

  if (uid == nexusUid) {
    currentCardName = "nexus";
    currentCardUid = uid;
    discountPercent = 0.0;
    discountAmount = 0.0;
    finalAmount = paymentAmount;
    return true;
  }

  if (uid == blazeUid) {
    currentCardName = "blaze";
    currentCardUid = uid;
    discountPercent = 0.0;
    discountAmount = 0.0;
    finalAmount = paymentAmount;
    return true;
  }

  return false;
}

bool checkPinForCurrentCard() {
  if (currentCardName == "flexi" && enteredPin == flexiPin) {
    return true;
  }

  if (currentCardName == "nexus" && enteredPin == nexusPin) {
    return true;
  }

  if (currentCardName == "blaze" && enteredPin == blazePin) {
    return true;
  }

  return false;
}

void returnToScanCardAfterWrongPin() {
  paymentStatus = "waiting_card";
  waitingForCard = true;
  waitingForPin = false;
  resetCardData();

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("WRONG PIN");
  lcd.setCursor(0, 1);
  lcd.print("SCAN AGAIN");

  delay(2000);

  showScanCard();
}

void setup() {
  Serial.begin(115200);

  SPI.begin();
  rfid.PCD_Init();

  Wire.begin(21, 22);

  lcd.init();
  lcd.backlight();

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("CONNECT WIFI");

  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("IP:");

  lcd.setCursor(0, 1);
  lcd.print(WiFi.localIP());

  Serial.println();
  Serial.print("ESP32 IP: ");
  Serial.println(WiFi.localIP());

  delay(3000);

  showReady();

  server.on("/start-payment", HTTP_POST, handleStartPayment);
  server.on("/payment-status", HTTP_GET, handlePaymentStatus);
  server.on("/cancel-payment", HTTP_POST, handleCancelPayment);

  server.begin();
}

void loop() {
  server.handleClient();

  if (waitingForCard) {

    if (rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) {

      String uid = readCardUid();

      Serial.print("CARD UID: ");
      Serial.println(uid);

      rfid.PICC_HaltA();
      rfid.PCD_StopCrypto1();

      if (selectCardByUid(uid)) {

        waitingForCard = false;
        waitingForPin = true;

        paymentStatus = "waiting_pin";

        enteredPin = "";

        lcd.clear();

        lcd.setCursor(0, 0);
        if (currentCardName == "flexi") {
          lcd.print("FLEXI PIN:");
        } else if (currentCardName == "nexus") {
          lcd.print("NEXUS PIN:");
        } else if (currentCardName == "blaze") {
          lcd.print("BLAZE PIN:");
        } else {
          lcd.print("ENTER PIN:");
        }

        lcd.setCursor(0, 1);

      } else {

        paymentStatus = "denied";
        waitingForCard = false;
        waitingForPin = false;
        resetCardData();

        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("INVALID CARD");

        delay(2500);

        showReady();
      }
    }
  }

  if (waitingForPin) {

    char key = keypad.getKey();

    if (key) {
      Serial.print("KEY: ");
      Serial.println(key);

      if (key >= '0' && key <= '9') {

        if (enteredPin.length() < 8) {

          enteredPin += key;

          lcd.setCursor(enteredPin.length() - 1, 1);
          lcd.print("*");
        }
      }

      if (key == '*') {

        enteredPin = "";

        lcd.setCursor(0, 1);
        lcd.print("                ");

        lcd.setCursor(0, 1);
      }

      if (key == '#') {

        waitingForPin = false;

        if (checkPinForCurrentCard()) {

          paymentStatus = "approved";

          lcd.clear();

          lcd.setCursor(0, 0);
          lcd.print("PAYMENT OK");

          lcd.setCursor(0, 1);
          if (discountPercent > 0) {
            lcd.print("-5% ");
          }
          lcd.print(finalAmount, 2);
          lcd.print(" EUR");

          delay(3000);

          showReady();

        } else {

          returnToScanCardAfterWrongPin();
        }
      }
    }
  }
}
